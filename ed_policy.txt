#Requires -Version 5.1
#Requires -RunAsAdministrator

Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

# ─────────────────────────────────────────────
#  CONFIGURATION
# ─────────────────────────────────────────────
$SourceAdmx    = '\\contoso\wind\sys\msedge.admx'
$SourceAdml    = '\\contoso\wind\sys\en-us\msedge.adml'
$DestAdmx      = 'C:\Windows\PolicyDefinitions'
$DestAdml      = 'C:\Windows\PolicyDefinitions\en-US'
$PolFile       = 'C:\Windows\System32\GroupPolicy\Machine\Registry.pol'

# ─────────────────────────────────────────────
#  HELPER – Status output
# ─────────────────────────────────────────────
function Write-Status {
    param([string]$Message, [string]$Color = 'Cyan')
    Write-Host "[$(Get-Date -Format 'HH:mm:ss')] $Message" -ForegroundColor $Color
}

# ─────────────────────────────────────────────
#  HELPER – Read Registry.pol into a byte array
#  Returns an empty (but valid) pol structure if
#  the file does not exist yet.
# ─────────────────────────────────────────────
function Get-PolBytes {
    param([string]$Path)
    # Registry.pol header: PReg signature (50 52 65 67) + version (01 00 00 00)
    $header = [byte[]](0x50,0x52,0x65,0x67,0x01,0x00,0x00,0x00)
    if (Test-Path $Path) {
        return [System.IO.File]::ReadAllBytes($Path)
    }
    return $header
}

# ─────────────────────────────────────────────
#  HELPER – Append one DWORD entry to pol bytes
#  Format per entry:
#    [  keypath  \0  valuename  \0 ; type ; size ; data  ]
#  All strings are UTF-16LE, integers are little-endian.
# ─────────────────────────────────────────────
function Add-PolDword {
    param(
        [byte[]]$PolBytes,
        [string]$KeyPath,
        [string]$ValueName,
        [uint32]$Value
    )

    $ms = New-Object System.IO.MemoryStream
    $ms.Write($PolBytes, 0, $PolBytes.Length)

    $w = New-Object System.IO.BinaryWriter($ms, [System.Text.Encoding]::Unicode)

    # Opening bracket
    $w.Write([uint16]0x005B)   # '['

    # Key path (UTF-16LE) + null terminator
    foreach ($c in $KeyPath.ToCharArray()) { $w.Write([uint16][char]$c) }
    $w.Write([uint16]0x0000)

    # Semicolon separator
    $w.Write([uint16]0x003B)   # ';'

    # Value name + null terminator
    foreach ($c in $ValueName.ToCharArray()) { $w.Write([uint16][char]$c) }
    $w.Write([uint16]0x0000)

    # Semicolon
    $w.Write([uint16]0x003B)

    # Type: REG_DWORD = 4  (little-endian DWORD)
    $w.Write([uint32]4)

    # Semicolon
    $w.Write([uint16]0x003B)

    # Size of data = 4 bytes
    $w.Write([uint32]4)

    # Semicolon
    $w.Write([uint16]0x003B)

    # Data value
    $w.Write([uint32]$Value)

    # Closing bracket
    $w.Write([uint16]0x005D)   # ']'

    $w.Flush()
    $result = $ms.ToArray()
    $w.Dispose()
    $ms.Dispose()
    return $result
}

# ─────────────────────────────────────────────
#  STEP 1 – Copy ADMX
# ─────────────────────────────────────────────
Write-Status "Copying msedge.admx..."
if (-not (Test-Path $SourceAdmx)) { Write-Error "Not found: $SourceAdmx"; exit 1 }
if (-not (Test-Path $DestAdmx))   { New-Item -Path $DestAdmx -ItemType Directory -Force | Out-Null }
Copy-Item -Path $SourceAdmx -Destination $DestAdmx -Force
Write-Status "msedge.admx -> $DestAdmx" 'Green'

# ─────────────────────────────────────────────
#  STEP 2 – Copy ADML
# ─────────────────────────────────────────────
Write-Status "Copying msedge.adml..."
if (-not (Test-Path $SourceAdml)) { Write-Error "Not found: $SourceAdml"; exit 1 }
if (-not (Test-Path $DestAdml))   { New-Item -Path $DestAdml -ItemType Directory -Force | Out-Null }
Copy-Item -Path $SourceAdml -Destination $DestAdml -Force
Write-Status "msedge.adml -> $DestAdml" 'Green'

# ─────────────────────────────────────────────
#  STEP 3 – Ensure GroupPolicy\Machine dir exists
# ─────────────────────────────────────────────
$polDir = Split-Path $PolFile
if (-not (Test-Path $polDir)) {
    New-Item -Path $polDir -ItemType Directory -Force | Out-Null
    Write-Status "Created: $polDir" 'Yellow'
}

# ─────────────────────────────────────────────
#  STEP 4 – Build / update Registry.pol
# ─────────────────────────────────────────────
$EdgeKey = 'Software\Policies\Microsoft\Edge'

Write-Status "Reading existing Registry.pol (or creating new)..."
[byte[]]$pol = Get-PolBytes -Path $PolFile

Write-Status "Writing 'EnableOnlineRevocationChecks = 0'..."
$pol = Add-PolDword -PolBytes $pol -KeyPath $EdgeKey `
                    -ValueName 'EnableOnlineRevocationChecks' -Value 0

Write-Status "Writing 'RequireOnlineRevocationChecksForLocalAnchors = 0'..."
$pol = Add-PolDword -PolBytes $pol -KeyPath $EdgeKey `
                    -ValueName 'RequireOnlineRevocationChecksForLocalAnchors' -Value 0

[System.IO.File]::WriteAllBytes($PolFile, $pol)
Write-Status "Registry.pol updated: $PolFile" 'Green'

# ─────────────────────────────────────────────
#  STEP 5 – Apply via gpupdate (no reboot needed)
# ─────────────────────────────────────────────
Write-Status "Running gpupdate /force..."
& gpupdate /force /target:computer
Write-Status "Done — policy applied without reboot." 'Green'
